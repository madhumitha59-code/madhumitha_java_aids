import java.util.*;
class main {
    public static void main(String[] args) {
        Scanner sc =new Scanner(System.in);
        int a[]=new int[5];
        a[0]=10;
        a[1]=20;
        a[2]=30;
        a[3]=40;
        int n=4;
        System.out.println("Enter Value:");
        int Value=sc.nextInt();
        System.out.println("Enter value(0 to"+ n + "):");
        int pos=sc.nextInt();
        if(pos<0||pos>n||n==a.length){
            System.out.println("Invalid Inserion");
        }else{
            for(int i=0;i<n;i++){
                a[i]=a[i-1];
            }
            a[pos]=Value;
            n++;
            for(int i=0;i<n;i++){
                System.out.print(a[i]+"");
            }
        }
        sc.close();
    }
}

class main {
    public static void main(String[] args) {
        int a[]={40,10,30,20,50};
        int key=20;
        int pos=-1;
        for(int i=0;i<a.length;i++){
            if(a[i]==key){
                pos=i;
                break;
            }
        }
        if(pos != -1)
            System.out.println("Found at index"+pos);
        else
            System.out.println("Not Found");
    }
}

class Main {
    public static void main(String[] args) {
    
        int a[]={10,20,30,40,50,60};
        int key=40;
        
        int low=0;
        int high=a.length-1;
        int pos =-1;
        
        while(low<=high){
            int mid=(low+high)/2;
            if(a[mid]==key){
                pos=mid;
                break;
            }else if(key>a[mid]){
                low=mid+1;
            }else{
                high=mid-1;
            }
        }
            if(pos!=-1){
                System.out.println("Found index"+pos);
            }else{
                System.out.println("Not Found");
            } 
        }
    }
